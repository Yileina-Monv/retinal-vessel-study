def J(p):return json.loads(Path(p).read_text(encoding='utf-8'))
def sha(p):
    h=hashlib.sha256()
    with Path(p).open('rb') as f:
        for b in iter(lambda:f.read(8*1024*1024),b''):h.update(b)
    return h.hexdigest()
def W(p,obj):
    with Path(p).open('x',encoding='utf-8') as f:json.dump(obj,f,ensure_ascii=False,indent=2,allow_nan=False)
def C(p,rows):
    with Path(p).open('x',newline='',encoding='utf-8-sig') as f:
        writer=csv.DictWriter(f,fieldnames=list(rows[0]));writer.writeheader();writer.writerows(rows)
def divide(n,d):return float(n/d) if d else None
def measure(pred,gt):
    sp=skeletonize(pred,method='zhang');sg=skeletonize(gt,method='zhang')
    pn=int(sp.sum());gn=int(sg.sum());pi=int((sp&gt).sum());gi=int((sg&pred).sum())
    tp=divide(pi,pn);ts=divide(gi,gn)
    if not pn or not gn:cl=1. if not pred.any() and not gt.any() else 0.
    else:cl=2*tp*ts/(tp+ts) if tp+ts else 0.
    return dict(topology_precision=tp,topology_sensitivity=ts,clDice=cl,
        pred_skeleton_denominator=pn,pred_skeleton_in_GT=pi,GT_skeleton_denominator=gn,GT_skeleton_in_pred=gi,
        false_pred_skeleton=pn-pi,empty_prediction=not pred.any(),empty_GT=not gt.any()),sp,sg
def tests():

    p=np.zeros_like(g);p[20:31,10:61]=1

    ok['width_blind_counterexample']=dict(clDice=r['clDice'],Dice=float(d))
    z=np.zeros_like(g);r,_,_=measure(z,z);assert r['clDice']==1 and r['topology_precision'] is None;ok['both_empty_explicit']=True
    r,_,_=measure(z,g);assert r['clDice']==0 and r['topology_sensitivity']==0;ok['prediction_empty']=True
    r,_,_=measure(g,z);assert r['clDice']==0 and r['topology_precision']==0;ok['GT_empty']=True
    r,_,_=measure(g,g);assert r['clDice']==1;ok['perfect']=True
    p=g.copy();p[5,5:15]=1;r,_,_=measure(p,g);assert r['topology_precision']<1 and r['topology_sensitivity']==1;ok['false_branch_penalized']=True
    p=g.copy();p[:,35]=0;r,_,_=measure(p,g);assert r['clDice']<1;ok['break_penalized']=dict(clDice=r['clDice'],pred_components=ndimage.label(p,np.ones((3,3)))[1])
    a,_,_=measure(g,p);assert a['clDice']==r['clDice'];ok['cldice_symmetric']=True
    q=np.zeros_like(g);q[2:4,2:4]=1;f=np.ones_like(g);f[:5]=0;assert not (q&f).any();ok['outside_FOV_zero_before_skeleton']=True
