def sha(p):
 p=Path(p);h=hashlib.sha256()
 with p.open('rb') as f:
  for b in iter(lambda:f.read(4194304),b''):h.update(b)
 return h.hexdigest()
def bind(p,expected=None):
 p=Path(p);h=sha(p)
 if expected is not None: assert h==expected,('source drift',str(p),h,expected)
 sources[str(p)]=dict(sha256=h,size=p.stat().st_size,mtime_ns=p.stat().st_mtime_ns)
 return p
