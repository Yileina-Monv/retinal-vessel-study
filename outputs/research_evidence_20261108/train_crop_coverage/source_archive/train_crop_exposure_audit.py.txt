INPUTS = {}
def sha(p):
    h=hashlib.sha256()
    with Path(p).open('rb') as f:
        for block in iter(lambda:f.read(1024*1024), b''): h.update(block)
    return h.hexdigest()
def pin(p, expected=None):
    p=Path(p); h=sha(p)
    if expected is not None and h!=expected: raise RuntimeError('Source changed: '+str(p))
    if str(p) in INPUTS and INPUTS[str(p)]!=h: raise RuntimeError('Concurrent source drift')
    INPUTS[str(p)]=h
    return p
def load(p):
    return json.loads(pin(p).read_text())
def rows(p):
    with pin(p).open(encoding='utf-8-sig', newline='') as f: return list(csv.DictReader(f))
def write_json(p, d):
    with Path(p).open('x', encoding='utf-8') as f: json.dump(d,f,ensure_ascii=False,indent=2,allow_nan=False)
def write_csv(p, d):
    with Path(p).open('x',encoding='utf-8',newline='') as f:
        w=csv.DictWriter(f,fieldnames=list(d[0]));w.writeheader();w.writerows(d)
def exposure(shape, rects, weights=None):
    h,w=shape; a=np.zeros((h+1,w+1),np.int32 if weights is None else np.float64)
    for i,(y,x,_,_,_) in enumerate(rects):
        v=1 if weights is None else weights[i]
        a[y,x]+=v;a[y+512,x]-=v;a[y,x+512]-=v;a[y+512,x+512]+=v
    return a.cumsum(0,dtype=a.dtype).cumsum(1,dtype=a.dtype)[:-1,:-1]
def expected_probability(h,w,patch=512):
    def axis(n):
        a=np.arange(n)
        return np.maximum(0,np.minimum(a,n-patch)-np.maximum(0,a-patch+1)+1)/(n-patch+1)
    return axis(h)[:,None]*axis(w)[None,:]
def integral_counts(mask, rects):
    a=np.pad(mask.astype(np.int64),((1,0),(1,0))).cumsum(0).cumsum(1)
    return [int(a[y+512,x+512]-a[y,x+512]-a[y+512,x]+a[y,x]) for y,x,*_ in rects]
def stats(c,weighted,expected,mask):
    a=c[mask];n=len(a)
    if not n:return dict(n=0,zero=None,zero_fraction=None,mean=None,p10=None,median=None,p90=None,min=None,max=None,weighted_mean=None,expected_mean=None,expected_zero_fraction=None)
    e=expected[mask]
    return dict(n=n,zero=int((a==0).sum()),zero_fraction=float((a==0).mean()),mean=float(a.mean()),p10=float(np.quantile(a,.1)),median=float(np.median(a)),p90=float(np.quantile(a,.9)),min=int(a.min()),max=int(a.max()),weighted_mean=float(weighted[mask].mean()),expected_mean=float(e.mean()),expected_zero_fraction=None)

def fixtures():
    shape=(520,530);rects=[(0,0,0,0,0),(8,18,1,1,3)];weights=[.2,.7]
    got=exposure(shape,rects); want=np.zeros(shape,np.int32);ww=np.zeros(shape)
    for row,weight in zip(rects,weights):
        y,x,*_=row;want[y:y+512,x:x+512]+=1;ww[y:y+512,x:x+512]+=weight
    assert np.array_equal(got,want) and np.allclose(exposure(shape,rects,weights),ww,rtol=0,atol=1e-15)
    ep=expected_probability(513,514)
    brute=np.zeros((513,514))
    for y in range(2):
        for x in range(3):brute[y:y+512,x:x+512]+=1/6
    assert np.allclose(ep,brute,rtol=0,atol=2e-16)
    assert abs(ep.sum()-512**2)<1e-8
    # Native crop membership remains unchanged by flips/90-degree rotations.
    mask=(np.arange(520*530).reshape(shape)%7==0)
    counts=integral_counts(mask,rects)
    for rect,count in zip(rects,counts):
        y,x,*_=rect;crop=mask[y:y+512,x:x+512]
        for hf in [False,True]:
            for vf in [False,True]:
                for rot in range(4):
                    z=crop[:,::-1] if hf else crop
                    z=z[::-1] if vf else z
                    assert int(np.rot90(z,rot).sum())==count
    return dict(rectangle_count_and_weight_exact=True,uniform_origin_probability_enumerated=True,all_16_transforms_preserve_native_membership=True)

def review_ufl():
    u=BASE/'ufl_asymmetric_v1_20261024/diagnostics'
    binding=load(u/'execution_binding.json');ledger=load(u/'diagnostic_ledger.json');review=load(u/'SOL_GPU_DIAGNOSTIC_REVIEW.json');completion=load(u/'completion.json')
    for p,h in review['evidence_sha256'].items():pin(p,h)
    assert ledger['status']=='COMPLETE' and ledger['forward_starts']==16 and ledger['autograd_calls']==96
    assert ledger['optimizer_step_attempts']==ledger['parameter_updates']==0
    assert len(ledger['entries'])==16 and all(x['status']=='COMPLETE' for x in ledger['entries'].values())
    vals=[];amp=[]
    for combo in binding['plan']:
        q=load(u/'attempt_01'/(combo['id']+'.json'));assert q['id']==combo['id'] and q['parameter_updates']==0
        assert q['unscaled'] and q['preclip'] and q['FOV_outside_gradient_zero']
        assert q['input_sha256']==binding['batch_input_sha256'][combo['batch_index']]
        for term,key in [('A0','A0_parameter_L2'),('UFL','candidate_parameter_L2'),('delta','delta_parameter_L2')]:
            assert math.isclose(math.sqrt(sum(v[term+'_L2']**2 for v in q['per_layer'].values())),q[key],rel_tol=1e-10)
        assert math.isclose(q['delta_to_A0'],q['delta_parameter_L2']/q['A0_parameter_L2'],rel_tol=1e-12)
        assert q['delta_to_A0']<=2 and q['GPU_peak_reserved_MiB']<=8192
        lim=binding['rules']['AMP_composite_relative_max' if combo['precision']=='AMP' else 'FP32_composite_relative_max']
        assert max(q['component_additivity_error'],q['difference_additivity_error'],q['scaled_backward_relative_error'])<=lim
        if q['AMP_vs_FP32']:
            a=q['AMP_vs_FP32'];assert a['loss_relative']<=.01 and a['total_gradient']['relative_L2']<=.05 and a['total_gradient']['cosine']>=.99;amp.append(a)
        vals.append(q)
    def span(key):return [min(q[key] for q in vals),max(q[key] for q in vals)]
    return dict(status='EXISTING_GPU16_ACCEPTANCE_INDEPENDENTLY_RECONCILED_CPU_ONLY',combinations=16,parameter_updates=0,delta_A0_ratio=span('delta_to_A0'),candidate_A0_ratio=span('candidate_to_A0'),cosine_A0_candidate=span('cosine_A0_candidate'),cosine_A0_delta=span('cosine_A0_delta'),max_AMP_gradient_relative_L2=max(a['total_gradient']['relative_L2'] for a in amp),min_AMP_gradient_cosine=min(a['total_gradient']['cosine'] for a in amp),max_reserved_MiB=completion['max_reserved_MiB'],elapsed_seconds=completion['elapsed_seconds'],prior_review_reused=True,new_GPU_calls=0,release_or_authorization_changed=False,limitation='Zero updates cannot establish training stability, restoration consistency, or Dice benefit. Delta is replacement UFL minus A0, not an auxiliary loss.')
