def sha(p):
    h=hashlib.sha256()
    with Path(p).open('rb') as f:
        for block in iter(lambda:f.read(1024*1024),b''):h.update(block)
    return h.hexdigest()
def load(p):return json.loads(Path(p).read_text())
def rows(p):
    with Path(p).open(encoding='utf-8-sig',newline='') as f:return list(csv.DictReader(f))
def jwrite(p,d):

def cwrite(p,data):

def origin_plan(height,width,n,image_index):
    assert n in (72,73) and height>=512 and width>=512
    rh,rw=height-511,width-511
    yedges=[k*rh//6 for k in range(7)];xedges=[k*rw//12 for k in range(13)]
    assert min(np.diff(yedges))>0 and min(np.diff(xedges))>0
    rng=np.random.default_rng(np.random.SeedSequence([2026,0x53504154,image_index]))
    grid=[]
    for gy in range(6):
        for gx in range(12):
            grid.append((int(rng.integers(yedges[gy],yedges[gy+1])),int(rng.integers(xedges[gx],xedges[gx+1])),gy,gx))
    grid=[grid[int(k)] for k in rng.permutation(72)]
    if n==73:grid.append((int(rng.integers(rh)),int(rng.integers(rw)),-1,-1))
    # Per-image draw marginal compared with the original uniform-origin law.
    ratio=np.array([rh*rw/(72*(yedges[y+1]-yedges[y])*(xedges[x+1]-xedges[x])) for y in range(6) for x in range(12)])
    if n==73:ratio=(72*ratio+1)/73
    return grid,dict(min_marginal_density_ratio=float(ratio.min()),max_marginal_density_ratio=float(ratio.max()),max_relative_deviation=float(abs(ratio-1).max()))

def summary(mask,count,weight):
    a=count[mask];n=len(a)
    return dict(n=n,zero=int((a==0).sum()),zero_fraction=float((a==0).mean()) if n else None,mean=float(a.mean()) if n else None,weighted_mean=float(weight[mask].mean()) if n else None)
