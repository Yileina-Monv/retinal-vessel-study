def sha(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as f:
        for b in iter(lambda:f.read(8*1024*1024), b''): h.update(b)
    return h.hexdigest()

def readj(p):
    return json.loads(Path(p).read_text(encoding='utf-8'))

def writej(p, data):
    with Path(p).open('x', encoding='utf-8') as f:
        json.dump(data, f, ensure_ascii=False, indent=2, allow_nan=False)
        f.write('\n')

def ratio(a,b):
    return float(a/b) if b else None

def stats(a):
    a = np.asarray(a)
    if not a.size: return dict(n=0, mean=None, median=None, q90=None, max=None, NA_reason='empty_set')
    return dict(n=int(a.size), mean=float(a.mean()), median=float(np.median(a)), q90=float(np.quantile(a,.9)), max=float(a.max()), NA_reason=None)

def check_prob(p,shape):
    if p.shape != shape or p.dtype != np.float32 or not np.isfinite(p).all() or np.any(p<0) or np.any(p>1):
        raise ValueError('invalid native float32 probability')

def fp_masks(probs,gt,fov):
    return {a:(p>=np.float32(.5)) & fov & ~gt for a,p in probs.items()}

def membership(fps):
    return fps['A0'].astype(np.uint8)+2*fps['B1'].astype(np.uint8)+4*fps['B2'].astype(np.uint8)

def sets_from_fp(fps):
    a,b,c = (fps[k] for k in METHODS)
    return dict(FP_A0=a,FP_B1=b,FP_B2=c,
        new_B1=b&~a,removed_B1=a&~b,retained_B1=a&b,
        new_B2=c&~a,removed_B2=a&~c,retained_B2=a&c,
        common_all=a&b&c,common_new=~a&b&c,
        exclusive_new_B1=~a&b&~c,exclusive_new_B2=~a&~b&c)

def tiles(shape):
    h,w=shape
    return [(x,y,min(x+512,w),min(y+512,h)) for y in range(0,h,512) for x in range(0,w,512)]

def best_tile(mask):
    candidates=[(int(mask[y:y1,x:x1].sum()),(x,y,x1,y1)) for x,y,x1,y1 in tiles(mask.shape)]
    score,box=max(candidates,key=lambda t:(t[0],-t[1][1],-t[1][0]))
    return box if score else None,score

def max_component(mask):
    lab,n=ndimage.label(mask,structure=np.ones((3,3),np.uint8))
    if not n: return None,0
    cnt=np.bincount(lab.ravel());cnt[0]=0;index=int(np.argmax(cnt))
    sl=ndimage.find_objects(lab==index)[0]
    y=(sl[0].start+sl[0].stop-1)//2; x=(sl[1].start+sl[1].stop-1)//2
    return (x//512*512,y//512*512,min((x//512+1)*512,mask.shape[1]),min((y//512+1)*512,mask.shape[0])),int(cnt[index])

def textfont(size=17):
    path=Path('/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf')
    return ImageFont.truetype(str(path),size) if path.exists() else ImageFont.load_default()

def paint(rgb, layers):
    z=rgb.astype(np.float32).copy()
    for mask,color,alpha in layers:
        z[mask]=(1-alpha)*z[mask]+alpha*np.asarray(color,np.float32)
    return Image.fromarray(z.clip(0,255).astype(np.uint8))

def panels(rgb,gt,fps,sets,code):
    result=[('RGB (unaltered)',Image.fromarray(rgb)),('GT: white',paint(rgb,[(gt,(255,255,255),.72)]))]
    for a in METHODS:
        pred_gt=gt & ~fps[a]
        # TP/FN are supplied separately by the caller; this placeholder is replaced.
        result.append((a,None))
    for a in ['B1','B2']:
        result.append((a+' vs A0: added RED / removed CYAN / retained YELLOW',paint(rgb,[(gt,(130,110,255),.23),
            (sets['retained_'+a],(255,230,30),.75),(sets['removed_'+a],(0,230,255),.85),(sets['new_'+a],(255,35,25),.90)])))
    result.append(('FP membership: bits A0=1 B1=2 B2=4; see HTML legend',paint(rgb,[(code==k,COLORS[k],.88) for k in range(1,8)])))
    return result

def save_sheet(path, items, title, crop=None, overview=False):
    size=410 if overview else 512
    h=round(size*items[0][1].height/items[0][1].width) if overview else (crop[3]-crop[1])
    w=size if overview else (crop[2]-crop[0])
    ch=h+47; cw=max(w,410)
    sheet=Image.new('RGB',(cw*4,ch*2+42),(17,20,26));d=ImageDraw.Draw(sheet)
    d.text((8,9),title,font=textfont(16),fill='white')
    for i,(label,im) in enumerate(items):
        a=im.resize((w,h),Image.Resampling.LANCZOS if i==0 else Image.Resampling.NEAREST) if overview else im.crop(crop)
        x=i%4*cw;y=i//4*ch+42
        sheet.paste(a,(x,y+42));d.text((x+5,y+3),label[:63],font=textfont(12),fill='white')
        if len(label)>63:d.text((x+5,y+19),label[63:126],font=textfont(12),fill='white')
    with Path(path).open('xb') as f: sheet.save(f,format='PNG')
    return sheet.size

def synth_tests():
    checks={}
    f=np.ones((7,7),bool);g=np.zeros_like(f);g[3,3]=1
    p={a:np.zeros((7,7),np.float32) for a in METHODS};p['B1'][3,4]=.5
    x=fp_masks(p,g,f);assert x['B1'][3,4] and not x['A0'].any();checks['threshold_equality']=True
    d=ndimage.distance_transform_edt(~g);assert d[3,4]==1 and d[4,4]>1;checks['native_euclidean_cross_not_diagonal']=True
    fps={a:np.array([(i&(1<<k))!=0 for i in range(8)])[None,:] for k,a in enumerate(METHODS)}
    assert np.array_equal(membership(fps),np.arange(8)[None,:]);checks['all_eight_memberships']=True
    ss=sets_from_fp(fps)
    for a in ['B1','B2']:
        assert int(ss['FP_'+a].sum())-int(ss['FP_A0'].sum())==int(ss['new_'+a].sum())-int(ss['removed_'+a].sum())
    checks['new_removed_identity']=True
    assert int(ss['common_all'].sum())==1 and int(ss['common_new'].sum())==1;checks['shared_not_double_counted']=True
    ff=f.copy();ff[3,4]=0;assert not fp_masks(p,g,ff)['B1'].any();checks['fov_exclusion']=True
    u=np.zeros_like(f);u[3,4]=1;assert (x['B1']&u).sum()==1;checks['U_retained_in_global_FP']=True
    z=np.eye(3,dtype=bool);assert ndimage.label(z,np.ones((3,3)))[1]==1;checks['eight_connected_diagonal']=True
    assert stats([])['mean'] is None and ratio(0,0) is None;checks['empty_NA']=True
    box,n=best_tile(np.zeros((520,520),bool));assert box is None and n==0;checks['empty_crop_NA']=True
    a=np.ones((1024,1024),bool);assert best_tile(a)[0]==(0,0,512,512);checks['crop_tie_row_column']=True
    assert max_component(np.zeros((5,5),bool))==(None,0);checks['empty_components']=True
    try:check_prob(np.ones((2,2),np.float64),(2,2))
    except ValueError:checks['reject_non_float32']=True
    else:raise AssertionError()
    assert 'torch' not in sys.modules;checks['no_torch_import']=True
    return checks
