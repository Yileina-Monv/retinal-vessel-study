"""Bind, preserve and aggregate the six actual per-image BCE contribution logs."""
import csv,hashlib,json,math,shutil
from collections import defaultdict
from pathlib import Path
ROOT=Path("${RETINA_PROJECT_ROOT}");OUT=ROOT/"experiments/t0_tw_weight_ablation_20261017";GROUPS=("A_V1","A_V0","O_V1","O_V0")
def sha(p):
 h=hashlib.sha256()
 with Path(p).open("rb") as f:
  for b in iter(lambda:f.read(1048576),b""):h.update(b)
 return h.hexdigest()
def writej(p,x):
 p=Path(p);p.parent.mkdir(parents=True,exist_ok=True);p.write_text(json.dumps(x,ensure_ascii=False,indent=2,allow_nan=False)+"\n",encoding="utf-8")
def wc(p,rows):
 if not rows:raise ValueError("Empty CSV")
 fields=list(dict.fromkeys(k for r in rows for k in r))
 with Path(p).open("w",newline="",encoding="utf-8-sig") as f:
  w=csv.DictWriter(f,fieldnames=fields,extrasaction="ignore");w.writeheader();w.writerows(rows)
def enrich_and_aggregate():
 protocol=json.loads((ROOT/"docs/t0_tw_weight_ablation_20261017/protocol.json").read_text())
 ver=json.loads((OUT/"implementation_gate/verification.json").read_text())
 if not ver.get("all_passed"):raise RuntimeError("Implementation gate missing")
 all_rows=[];audit=[]
 for seed in (2026,2027,2028):
  for alpha,label in ((1.5,"1p5"),(3.0,"3p0")):
   run=OUT/f"runs/seed_{seed}/w_{label}";summary=json.loads((run/"summary.json").read_text())
   if summary.get("status")!="passed" or summary.get("completed_steps")!=2000 or summary.get("next_global_draw")!=8000:raise RuntimeError("Incomplete track "+str(run))
   src=run/"bce_contributions.jsonl";raw=run/"bce_contributions.raw.jsonl";bind=run/"run_binding.json"
   if not raw.exists():shutil.copy2(src,raw)
   runbind=json.loads(bind.read_text());runbindsha=sha(bind);out=[];prehash=sha(raw)
   with raw.open(encoding="utf-8") as f:
    for line in f:
     row=json.loads(line)
     row.update(protocol_sha256=protocol["protocol_id"],protocol_file_sha256=ver["protocol_sha256"],
       execution_binding_sha256=ver["execution_binding_sha256"],run_binding_sha256=runbindsha,
       seed=seed,alpha=alpha)
     out.append(row)
   tmp=src.with_suffix(".enriched.tmp")
   with tmp.open("w",encoding="utf-8") as f:
    for row in out:f.write(json.dumps(row,ensure_ascii=False,allow_nan=False)+"\n")
   tmp.replace(src)
   if len(out)!=8000:raise RuntimeError("Contribution row count differs")
   audit.append({"seed":seed,"alpha":alpha,"raw_path":str(raw),"raw_sha256":prehash,"enriched_path":str(src),"enriched_sha256":sha(src),"rows":len(out),"run_binding_sha256":runbindsha})
   all_rows.extend(out)
 if len(all_rows)!=48000:raise RuntimeError("Expected 48000 new contribution rows")
 summary_rows=[];window_rows=[];image_rows=[]
 for seed in (2026,2027,2028):
  for alpha in (1.5,3.0):
   rows=[r for r in all_rows if r["seed"]==seed and r["alpha"]==alpha]
   key={"seed":seed,"alpha":alpha};mean_bce=sum(r["BCE_image"] for r in rows)/len(rows);sum_f=sum(r["nF"] for r in rows);sum_a=sum(r["nA"] for r in rows)
   zero_a=sum(r["nA"]==0 for r in rows)
   for g in GROUPS:
    n=sum(r["groups"][g]["n"] for r in rows);mean_c=sum(r["groups"][g]["C"] for r in rows)/len(rows)
    mean_s=sum(r["groups"][g]["S"] for r in rows)/len(rows);mean_w=sum(r["groups"][g]["W"] for r in rows)/len(rows)
    summary_rows.append({**key,"group":g,"draws":len(rows),"mean_n_per_draw":n/len(rows),"sampled_pixel_instances":n,
      "sampled_pixel_share_of_FOV":n/sum_f if sum_f else None,"mean_raw_BCE_sum_S":mean_s,"mean_weighted_BCE_sum_W":mean_w,
      "mean_normalized_contribution_C":mean_c,"ratio_of_means_C_over_BCE":mean_c/mean_bce if mean_bce else None,
      "mean_BCE_image":mean_bce,"zero_A_draws":zero_a,"nonzero_A_draws":len(rows)-zero_a})
   for lo in range(1,2001,200):
    win=[r for r in rows if lo<=r["step"]<lo+200];mb=sum(r["BCE_image"] for r in win)/len(win)
    for g in GROUPS:
     n=sum(r["groups"][g]["n"] for r in win);mc=sum(r["groups"][g]["C"] for r in win)/len(win)
     window_rows.append({**key,"step_first":lo,"step_last":lo+199,"draws":len(win),"group":g,"sampled_pixel_instances":n,
      "sampled_pixel_share_of_FOV":n/sum(r["nF"] for r in win) if win else None,
      "mean_normalized_contribution_C":mc,"mean_BCE_image":mb,"ratio_of_means_C_over_BCE":mc/mb if mb else None,
      "zero_A_draws":sum(r["nA"]==0 for r in win)})
   byname=defaultdict(list)
   for r in rows:byname[r["filename"]].append(r)
   for name,items in byname.items():
    for g in GROUPS:
     image_rows.append({**key,"filename":name,"draws":len(items),"group":g,
       "mean_C":sum(r["groups"][g]["C"] for r in items)/len(items),
       "mean_S":sum(r["groups"][g]["S"] for r in items)/len(items),
       "sampled_pixel_instances":sum(r["groups"][g]["n"] for r in items)})
 wc(OUT/"analysis/bce_contribution_summary.csv",summary_rows)
 wc(OUT/"analysis/bce_contributions_by_window.csv",window_rows)
 wc(OUT/"analysis/bce_contributions_by_image.csv",image_rows)
 writej(OUT/"analysis/bce_contribution_log_audit.json",{"status":"passed","rows_total":len(all_rows),"expected_rows_total":48000,"per_run":audit,
  "definitions":"Per-image normalized C averaged over 8000 sampled image-draws; contribution shares use ratio-of-means; pixel counts are sampled pixel instances with repeated crops, not unique pixels. Old alpha 1/2 contribution logs unavailable and remain NA."})
 return audit
if __name__=="__main__":print(json.dumps(enrich_and_aggregate(),ensure_ascii=False,indent=2))
