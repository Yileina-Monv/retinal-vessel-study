def sha(p):
 h=hashlib.sha256()
 with Path(p).open("rb") as f:
  for b in iter(lambda:f.read(1048576),b""):h.update(b)
 return h.hexdigest()
def wj(p,x):

def wc(p,rows):

  w=csv.DictWriter(f,fieldnames=keys,extrasaction="ignore");w.writeheader();w.writerows(rows)

def paired_statistics(vals,valid):
 paired_rows=[];seed_rows=[];boot_rows=[];stats={};rngs={}
 for m in METRICS:
  ids=valid[m];n=len(ids)
  if n not in rngs:rngs[n]=np.random.Generator(np.random.PCG64(20261016)).integers(0,n,(1000,n),dtype=np.int64)
  ix=rngs[n]
  for a,b in CONTRASTS:
   mat=np.empty((n,3),dtype=np.float64)
   for j,s in enumerate(SEEDS):
    av=np.array([vals[(a,s)][name][m]["value"] for name in ids],dtype=float)
    bv=np.array([vals[(b,s)][name][m]["value"] for name in ids],dtype=float)
    if not np.isfinite(av).all() or not np.isfinite(bv).all():raise RuntimeError(f"NA in frozen valid set {m}/{a}-{b}/{s}")
    mat[:,j]=av-bv
    seed_rows.append({"metric":m,"contrast":f"{a:g}-{b:g}","alpha":a,"reference_alpha":b,"seed":s,"valid_n":n,
       "alpha_macro":float(av.mean()),"reference_macro":float(bv.mean()),"delta_macro":float((av-bv).mean())})
   mean_by_image=mat.mean(axis=1);estimate=float(mean_by_image.mean())
   be=mean_by_image[ix].mean(axis=1);lo,hi=np.quantile(be,[.025,.975],method="linear")
   sd=float(mat.mean(axis=0).std(ddof=1))
   key=f"{a:g}-{b:g}"
   stats.setdefault(key,{})[m]={"estimate":estimate,"ci95_low":float(lo),"ci95_high":float(hi),"valid_n":n,
      "seed_delta_mean":float(mat.mean(axis=0).mean()),"seed_delta_sd":sd,
      "seed_deltas":{str(s):float(mat[:,j].mean()) for j,s in enumerate(SEEDS)}}
   for i,name in enumerate(ids):
    paired_rows.append({"metric":m,"contrast":key,"filename":name,"delta_mean3seed":float(mean_by_image[i]),
      **{f"delta_seed_{s}":float(mat[i,j]) for j,s in enumerate(SEEDS)}})
   for k,estimate_b in enumerate(be):
    boot_rows.append({"metric":m,"contrast":key,"draw":k+1,"indices_json":json.dumps(ix[k].tolist()),"estimate":float(estimate_b)})
 return stats,paired_rows,seed_rows,boot_rows
def z0_comparison(vals,z0,valid):
 rows=[];summary=[]
 for m,ids in valid.items():
  zbase=float(np.mean([z0[n][m]["value"] for n in ids]))
  for a in ALPHAS:
   seedvals=[]
   for s in SEEDS:
    v=float(np.mean([vals[(a,s)][n][m]["value"] for n in ids]));seedvals.append(v)
    rows.append({"metric":m,"alpha":a,"seed":s,"valid_n":len(ids),"Z0_macro":zbase,"endpoint_macro":v,"endpoint_minus_Z0":v-zbase})
   summary.append({"metric":m,"alpha":a,"valid_n":len(ids),"Z0_macro":zbase,"endpoint_macro_mean3seed":float(np.mean(seedvals)),
      "endpoint_minus_Z0_mean3seed":float(np.mean(seedvals)-zbase),**{f"seed_{s}_macro":seedvals[i] for i,s in enumerate(SEEDS)}})
 return rows,summary
def same_fp(probs,refs,z0,bind):
 target=float(np.mean([z0[n]["fp_density"]["value"] for n in refs]))
 gridrows=[];same={};selected=[]
 for a in ALPHAS:
  for s in SEEDS:
   counts=[];fovs=[]
   for n,r in refs.items():
    p=probs[(a,s)][n];v=load_mask(r["vessel_path"]);f=load_mask(r["fov_path"])
    counts.append(probability_fp_grid(p,v,f,GRID));fovs.append(int(f.sum()))
   dens=np.mean(np.stack(counts)/np.asarray(fovs)[:,None],axis=0);ok=np.flatnonzero(dens<=target);ix=int(ok[0]) if len(ok) else None
   same[(a,s)]={"threshold":float(GRID[ix]) if ix is not None else None,"grid_index":ix,"macro_fp_density":float(dens[ix]) if ix is not None else None,"target":target}
   for j,t in enumerate(GRID):gridrows.append({"alpha":a,"seed":s,"grid_index":j,"threshold":float(t),"macro_fp_density":float(dens[j]),"target":target,"feasible":int(dens[j]<=target)})
   if ix is not None:
    for n,r in refs.items():
     q=eval_prob(probs[(a,s)][n],r,float(GRID[ix]))
     selected.append({"alpha":a,"seed":s,"filename":n,"threshold":float(GRID[ix]),**{m:q[m]["value"] for m in METRICS},
       **{m+"_numerator":q[m]["numerator"] for m in METRICS},**{m+"_denominator":q[m]["denominator"] for m in METRICS},
       **{m+"_NA_reason":q[m]["NA_reason"] for m in METRICS}})
 return target,same,gridrows,selected
